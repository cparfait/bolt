# À copier dans /var/www/html/robots-boltpointage.txt sur l'Apache en DMZ
# (voir deploy/apache-emargement.conf, Alias /robots.txt).
#
# Ce fichier ne protège rien : un robot hostile l'ignore, et le cloisonnement
# reste assuré par le « Require all denied » du vhost. Il évite seulement que
# la feuille d'émargement et l'espace agent remontent dans un moteur de
# recherche — et il donne à Apache autre chose qu'un 403 à répondre, ce qui
# évite de nourrir les scénarios CrowdSec avec du trafic parfaitement normal.
User-agent: *
Disallow: /
